 154: float ASimpleRadar::ComputeRadarSignalStrength(const FHitResult& Hit)
 155: {
 156: 	// Frequenz 77GHz, Wellenlï¿½nge= 3.9mm, Sendeleistung 0.01W, Antennengewinn 30dB ca. 1000(linear), Radarziel Rï¿½ckstreuquerschnitt (Auto) 10m2, R = Entfernung zum Ziel
 157: 	// Pr = (7.5 * 10^-5 * Radarziel Rï¿½ckstreuquerschnitt) / R^4
 158: 	float backscatterValue = 10; // default for cars, need adjustment for bikes and pedestrians
 159: 
 160: 	float distanceToTarget = (Hit.ImpactPoint - Hit.TraceStart).Size() * 0.01; //in meter
 161: 	distanceToTarget = FMath::Max(distanceToTarget, 0.01f); // prevent division by zero
 162: 	float signalStrengthInWatt = (7.5 * 0.00001 * backscatterValue) / FMath::Pow(distanceToTarget, 4); 
 163: 
 164: 	float signalStrengthInDB = 10 * FMath::LogX(10, (signalStrengthInWatt / FMath::Pow(10.0, -12.0)));
 165: 
 166: 	// normalise dB signal
 167: 	float scaledDBsignal = (signalStrengthInDB) / 100.0;
 168: 
 169: 	return scaledDBsignal;
 170: }
