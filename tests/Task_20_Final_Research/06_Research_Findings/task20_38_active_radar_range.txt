 174: 	Sensor.SensorViewConfig->clear_radar_sensor_view_configuration();
 175: 	auto* conf = Sensor.SensorViewConfig->add_radar_sensor_view_configuration();
 176: 	conf->set_field_of_view_horizontal(SensorConfiguration.Radar.FieldOfViewHorizontal);
 177: 	conf->set_field_of_view_vertical(SensorConfiguration.Radar.FieldOfViewVertical);
 178: 	conf->set_number_of_rays_horizontal(SensorConfiguration.Radar.NumberOfRaysHorizontal);
 179: 	conf->set_number_of_rays_vertical(SensorConfiguration.Radar.NumberOfRaysVertical);
 180: 	conf->mutable_mounting_position()->set_allocated_position(
 181: 		ToOSI::Position(SensorConfiguration.Offset.GetLocation()));
 182: 	conf->mutable_mounting_position()->set_allocated_orientation(

 184: 	conf->set_emitter_frequency(77000000000.0); //in Hz
 185: 	//calculate angles
 186: 
 187: 	Sensor.AzimuthAngles.Empty();
