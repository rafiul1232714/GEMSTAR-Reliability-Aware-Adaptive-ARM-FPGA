 182: void ARadarSensorManager::SpawnSensor(FSensorConfigData& sensorConfiguration) {
 183: 	UWorld* World = GetWorld();
 184: 	if (World)
 185: 	{
 186: 		FActorSpawnParameters params;
 187: 		params.Name = sensorConfiguration.ID;
 188: 		ASimpleRadar* SpawnedActor = World->SpawnActor<ASimpleRadar>(ASimpleRadar::StaticClass(), FTransform(), params);
 189: 		if (SpawnedActor == nullptr) {
 190: 			UE_LOG(Radar, Log, TEXT("RadarSensorManager: Radar spawn failed for %s"), *SpawnedActor->GetName());
 191: 			return;
 192: 		}
 193: 
 194: 		SpawnedActor->SetSensorConfiguration(sensorConfiguration);
 195: 
 196: 		UE_LOG(Radar, Log, TEXT("RadarSensorManager: Radar spawned %s"), *SpawnedActor->GetName());
 197: 		if (!sensorConfiguration.Mount.IsEmpty()) {
 198: 			IGemstarManager::Execute_MountOn(GemstarSimulationSubsystem, sensorConfiguration.Mount, SpawnedActor);
 199: 		}
 200: 		else {
 201: 			ILocalTransformUpdater::Execute_UpdateTransformAfterAttachment(SpawnedActor);
 202: 		}
 203: 		AddTickPrerequisiteActor(SpawnedActor);
 204: 		SpawnedActors.Add(SpawnedActor);
 205: 	}

 208: void ARadarSensorManager::SpawnSensor_Implementation(FSensorConfigData& sensorConfiguration)
 209: {
 210: 	if (sensorConfiguration.Type != SensorType::RADAR) {
 211: 		return;
 212: 	}
 213: 
 214: 	UWorld* World = GetWorld();
 215: 	if (World)
 216: 	{
 217: 		FActorSpawnParameters params;
 218: 		if (sensorConfiguration.ID.IsNone()) {
 219: 			params.Name = "Unnamed_Radar_Sensor";
 220: 		}
 221: 		else {
 222: 			params.Name = sensorConfiguration.ID;
 223: 		}
 224: 		params.Owner = sensorConfiguration.Vehicle;
 225: 
 226: 		ASimpleRadar* SpawnedActor = World->SpawnActor<ASimpleRadar>(ASimpleRadar::StaticClass(), FTransform(), params);
 227: 		SpawnedActor->AttachToActor(sensorConfiguration.Vehicle, FAttachmentTransformRules(EAttachmentRule::SnapToTarget, true));
 228: 		SpawnedActor->SetSensorConfiguration(sensorConfiguration);
 229: 		SpawnedActor->SetActorRelativeTransform(sensorConfiguration.Offset);
 230: 
 231: 		if (SpawnedActor)
 232: 		{
 233: 			UE_LOG(Radar, Warning, TEXT("RadarSensorManager: spawned radar sensor: %s"), *SpawnedActor->GetName());
 234: 			SpawnedActors.Add(SpawnedActor);
 235: 		}
 236: 	}
