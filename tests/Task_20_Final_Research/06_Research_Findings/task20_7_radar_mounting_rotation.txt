  56: 	// Get owner's location and rotation
  57: 	const FVector RadarPosition = this->GetActorLocation();
  58: 	const FRotator RadarRotation = this->GetActorRotation();

  60: 	// Initialize array for raycast result

  62: 
  63: 	// Calculate batch size for 'ParallelFor' based on workable thread
  64: 	const int32 ThreadNum = FPlatformMisc::NumberOfWorkerThreadsToSpawn();
  65: 	const int32 DivideEnd = FMath::FloorToInt((float)(Sensor.RecordedHits.Num() / ThreadNum));
  66: 
  67: 	ParallelFor(
  68: 		ThreadNum,
  69: 		[&](int32 PFIndex)
  70: 		{
  71: 			int32 StartAt = PFIndex * DivideEnd;
  72: 			if (StartAt >= Sensor.RecordedHits.Num()) {
  73: 				return;
  74: 			}
  75: 
  76: 			int32 EndAt = StartAt + DivideEnd;
  77: 			if (PFIndex == (ThreadNum - 1)) {
  78: 				EndAt = Sensor.RecordedHits.Num();
  79: 			}
  80: 
  81: 			for (int32 Index = StartAt; Index < EndAt; ++Index) {
  82: 				const float HAngle = Sensor.AzimuthAngles[Index % Sensor.Number_of_rays_horizontal];
  83: 				const float VAngle = Sensor.ElevationAngles[Index / Sensor.Number_of_rays_horizontal];

  88: 				FVector EndPoint = RadarPosition + Sensor.SensorViewConfig->range() * 100 * UKismetMathLibrary::GetForwardVector(Rotation);

  90: 				GetWorld()->LineTraceSingleByChannel(
  91: 					Sensor.RecordedHits[Index].HitResult, RadarPosition, EndPoint, ECC_Visibility, TraceParams, FCollisionResponseParams::DefaultResponseParam);
