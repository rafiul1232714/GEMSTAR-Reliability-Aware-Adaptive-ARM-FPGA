  18: void ASimpleRadar::BeginPlay()
  19: {
  20: 	Super::BeginPlay();
  21: 	  if (Sensor.SensorViewConfig == nullptr)
  22:   {
  23:       CreateTestConfig();
  24:   }
  25: 	
  26: 	//initialise internal speed calculation
  27: 	currentVelocity = FVector::ZeroVector;
  28: 	oldPosition = GetActorLocation();
  29: 	// complex collisions: true
  30: 	TraceParams = FCollisionQueryParams(TEXT("RadioTrace"), true, GetOwner());
  31: 	TraceParams.bReturnPhysicalMaterial = true;
  32: 	TraceParams.bTraceComplex = true;
  33: }

  35: // Called every frame
  36: void ASimpleRadar::Tick(float DeltaTime){
  37: 	if (!TickActive) return;
  38: 	//internal speed calculation
  39: 	if (firstTick) {
  40: 		firstTick = false;
  41: 		currentVelocity = FVector::ZeroVector;
  42: 	} else {
  43: 		currentVelocity = (GetActorLocation() - oldPosition) / DeltaTime;
  44: 		oldPosition = GetActorLocation();
  45: 	}
  46: 
  47: 	GetScanData();
  48: 	if (bDebugRayTraces || bDebugReflections || bDebugNormals)
  49: 		ShowSensorDebugInformation();
  50: 	if (PublishingActive)
  51: 	{
  52: 		GenerateOSIMessage();
  53: 		SendData(*Sensor.SensorView);
  54: 	}
  55: }

  57: void ASimpleRadar::GetScanData() {
  58: 
  59: 	// Get owner's location and rotation
  60: 	const FVector RadarPosition = this->GetActorLocation();
  61: 	const FRotator RadarRotation = this->GetActorRotation();
  62: 
  63: 	// Initialize array for raycast result
  64: 	Sensor.RecordedHits.Init(FRadarHitResultAndIntesity(), Sensor.Number_of_rays_horizontal * Sensor.Number_of_rays_vertical);
  65: 
  66: 	// Calculate batch size for 'ParallelFor' based on workable thread
  67: 	const int32 ThreadNum = FPlatformMisc::NumberOfWorkerThreadsToSpawn();
  68: 	const int32 DivideEnd = FMath::FloorToInt((float)(Sensor.RecordedHits.Num() / ThreadNum));
  69: 
  70: 	ParallelFor(
  71: 		ThreadNum,
  72: 		[&](int32 PFIndex)
  73: 		{
  74: 			int32 StartAt = PFIndex * DivideEnd;
  75: 			if (StartAt >= Sensor.RecordedHits.Num()) {
  76: 				return;
  77: 			}
  78: 
  79: 			int32 EndAt = StartAt + DivideEnd;
  80: 			if (PFIndex == (ThreadNum - 1)) {
  81: 				EndAt = Sensor.RecordedHits.Num();
  82: 			}
  83: 
  84: 			for (int32 Index = StartAt; Index < EndAt; ++Index) {
  85: 				const float HAngle = Sensor.AzimuthAngles[Index % Sensor.Number_of_rays_horizontal];
  86: 				const float VAngle = Sensor.ElevationAngles[Index / Sensor.Number_of_rays_horizontal];
  87: 
  88: 				FRotator RadioRotation(FMath::RadiansToDegrees(VAngle), FMath::RadiansToDegrees(HAngle), 0.f);
  89: 				FRotator Rotation = UKismetMathLibrary::ComposeRotators(RadioRotation, RadarRotation);
  90: 
  91: 				FVector EndPoint = RadarPosition + Sensor.SensorViewConfig->range() * 100 * UKismetMathLibrary::GetForwardVector(Rotation);
  92: 
  93: 				GetWorld()->LineTraceSingleByChannel(
  94: 					Sensor.RecordedHits[Index].HitResult, RadarPosition, EndPoint, ECC_Visibility, TraceParams, FCollisionResponseParams::DefaultResponseParam);
  95: 				if (Sensor.RecordedHits[Index].HitResult.bBlockingHit) {
  96: 					Sensor.RecordedHits[Index].Intesity = ComputeRadarSignalStrength(Sensor.RecordedHits[Index].HitResult);
  97: 				}
  98: 			}
  99: 		},
 100: 		!SupportMultithread
 101: 	);
 102: 
 103: }

 105: void ASimpleRadar::GenerateOSIMessage() {
 106: 	delete Sensor.SensorView;
 107: 	Sensor.SensorView = new osi3::SensorView();
 108: 	auto* radarSensorView = Sensor.SensorView->add_radar_sensor_view();
 109: 	auto* viewConfiguration = radarSensorView->mutable_view_configuration();
 110: 	if (Sensor.SensorViewConfig != nullptr && Sensor.SensorViewConfig->radar_sensor_view_configuration_size())
 111: 		viewConfiguration->CopyFrom(Sensor.SensorViewConfig->radar_sensor_view_configuration(0));
 112: 
 113: 	
 114: 	for (int Index = 0; Index != Sensor.RecordedHits.Num(); Index++) {
 115: 		FRadarHitResultAndIntesity hit = Sensor.RecordedHits[Index];
 116: 		auto* reflection = radarSensorView->add_reflection();
 117: 		
 118: 		reflection->set_time_of_flight(2 * 0.01 * hit.HitResult.Distance / SPEED_OF_LIGHT);
 119: 		reflection->set_source_horizontal_angle(Sensor.AzimuthAngles[Index % Sensor.Number_of_rays_horizontal]);
 120: 		reflection->set_source_vertical_angle(Sensor.ElevationAngles[Index / Sensor.Number_of_rays_horizontal]);
 121: 		reflection->set_doppler_shift(0);
 122: 		reflection->set_signal_strength(hit.Intesity);
 123: 		FVector hitVelocity = FVector::ZeroVector;
 124: 		AActor* hitTarget = hit.HitResult.GetActor();
 125: 		if (hit.HitResult.bBlockingHit) {
 126: 			if (hitTarget->Implements<USensorReflectionData>()) {
 127: 				hitVelocity = Cast<ISensorReflectionData>(hitTarget)->GetVelocity();
 128: 			}
 129: 			else {
 130: 				hitVelocity = hitTarget->GetVelocity();
 131: 			}
 132: 			FString hitName = hit.HitResult.GetHitObjectHandle().GetName();
 133: 			FVector relativePostition = hit.HitResult.ImpactPoint - hit.HitResult.TraceStart;
 134: 			FVector relativeVelocity = hitVelocity - currentVelocity;
 135: 			float distance = relativePostition.Size();
 136: 			float radialVelocity = FVector::DotProduct(relativeVelocity, relativePostition / distance);
 137: 			double frequency = viewConfiguration->emitter_frequency();
 138: 			double doppler_shift = 2 * (radialVelocity * 0.01) * frequency / SPEED_OF_LIGHT;
 139: 			reflection->set_doppler_shift(doppler_shift);
 140: 			//Debug Data
 141: 			//if (hitName.Contains("BP_Simple")) {UE_LOG(Radar, Log, TEXT("Hit_%d: %s, vHero= %s, vTarget= %s, vR= %.2f, dS= %.2f, sS= %.2f" ), Index, *hitName, *currentVelocity.ToCompactString(), *hitVelocity.ToCompactString(), radialVelocity, doppler_shift, hit.Intesity);}
 142: 		}
 143: 	}
 144: 	
 145: 	Parent = GetAttachParentActor();
 146: 
 147: 	if (Parent != nullptr) {
 148: 		auto* moving_object = Sensor.SensorView->mutable_global_ground_truth()->add_moving_object();
 149: 		uint32_t uniqueID = Parent->GetUniqueID();
 150: 		moving_object->mutable_id()->set_value(uniqueID);
 151: 		moving_object->mutable_base()->set_allocated_position(ToOSI::Position(Parent->GetActorLocation()));
 152: 		moving_object->mutable_base()->set_allocated_orientation(ToOSI::Orientation3d(Parent->GetActorQuat()));
 153: 		Sensor.SensorView->mutable_global_ground_truth()->mutable_host_vehicle_id()->set_value(uniqueID);
 154: 	}
 155: }

 157: float ASimpleRadar::ComputeRadarSignalStrength(const FHitResult& Hit)
 158: {
 159: 	// Frequenz 77GHz, Wellenlï¿½nge= 3.9mm, Sendeleistung 0.01W, Antennengewinn 30dB ca. 1000(linear), Radarziel Rï¿½ckstreuquerschnitt (Auto) 10m2, R = Entfernung zum Ziel
 160: 	// Pr = (7.5 * 10^-5 * Radarziel Rï¿½ckstreuquerschnitt) / R^4
 161: 	float backscatterValue = 10; // default for cars, need adjustment for bikes and pedestrians
 162: 
 163: 	float distanceToTarget = (Hit.ImpactPoint - Hit.TraceStart).Size() * 0.01; //in meter
 164: 	distanceToTarget = FMath::Max(distanceToTarget, 0.01f); // prevent division by zero
 165: 	float signalStrengthInWatt = (7.5 * 0.00001 * backscatterValue) / FMath::Pow(distanceToTarget, 4); 
 166: 
 167: 	float signalStrengthInDB = 10 * FMath::LogX(10, (signalStrengthInWatt / FMath::Pow(10.0, -12.0)));
 168: 
 169: 	// normalise dB signal
 170: 	float scaledDBsignal = (signalStrengthInDB) / 100.0;
 171: 
 172: 	return scaledDBsignal;
 173: }

 175: void ASimpleRadar::SetSensorConfiguration(FSensorConfigData NewSensorConfiguration) {
 176: 	Super::SetSensorConfiguration(NewSensorConfiguration);
 177:   if (Sensor.SensorViewConfig == nullptr)
 178:   {
 179:       CreateTestConfig();
 180:   }
 181:   Sensor.SensorViewConfig->set_range(SensorConfiguration.Radar.Range);
 182:   UE_LOG(LogTemp, Warning,
 183:       TEXT("TASK20_45_RADAR_RANGE: %.3f metres"),
 184:       Sensor.SensorViewConfig->range());
 185: 	Sensor.SensorViewConfig->clear_radar_sensor_view_configuration();
 186: 	auto* conf = Sensor.SensorViewConfig->add_radar_sensor_view_configuration();
 187: 	conf->set_field_of_view_horizontal(SensorConfiguration.Radar.FieldOfViewHorizontal);
 188: 	conf->set_field_of_view_vertical(SensorConfiguration.Radar.FieldOfViewVertical);
 189: 	conf->set_number_of_rays_horizontal(SensorConfiguration.Radar.NumberOfRaysHorizontal);
 190: 	conf->set_number_of_rays_vertical(SensorConfiguration.Radar.NumberOfRaysVertical);
 191: 	conf->mutable_mounting_position()->set_allocated_position(
 192: 		ToOSI::Position(SensorConfiguration.Offset.GetLocation()));
 193: 	conf->mutable_mounting_position()->set_allocated_orientation(
 194: 		ToOSI::Orientation3d(SensorConfiguration.Offset.GetRotation()));
 195: 	conf->set_emitter_frequency(77000000000.0); //in Hz
 196: 	//calculate angles
 197: 
 198: 	Sensor.AzimuthAngles.Empty();
 199: 	Sensor.ElevationAngles.Empty();
 200: 	Sensor.Number_of_rays_horizontal = Sensor.SensorViewConfig->radar_sensor_view_configuration(0).number_of_rays_horizontal();
 201: 	for (uint32_t i = 0; i < Sensor.Number_of_rays_horizontal; i++) {
 202: 		Sensor.AzimuthAngles.Add(-0.5 * Sensor.SensorViewConfig->radar_sensor_view_configuration(0).field_of_view_horizontal()
 203: 			+ Sensor.SensorViewConfig->radar_sensor_view_configuration(0).field_of_view_horizontal() / (Sensor.Number_of_rays_horizontal - 1) * i);
 204: 	}
 205: 	Sensor.Number_of_rays_vertical = Sensor.SensorViewConfig->radar_sensor_view_configuration(0).number_of_rays_vertical();
 206: 	for (uint32_t i = 0; i < Sensor.Number_of_rays_vertical; i++) {
 207: 		Sensor.ElevationAngles.Add(-0.5 * Sensor.SensorViewConfig->radar_sensor_view_configuration(0).field_of_view_vertical()
 208: 			+ Sensor.SensorViewConfig->radar_sensor_view_configuration(0).field_of_view_vertical() / (Sensor.Number_of_rays_vertical - 1) * i);
 209: 	}
 210: }

 217: void ASimpleRadar::ShowSensorDebugInformation() {
 218: 	int noHitCounter = 0;
 219: 	for (auto& hit : Sensor.RecordedHits) {
 220: 		if (hit.HitResult.bBlockingHit) {
 221: 			if (bDebugRayTraces)
 222: 				DrawDebugLine(GetWorld(), hit.HitResult.TraceStart, hit.HitResult.ImpactPoint, FColor::Green, false, 0.0f, 0, 1.0f);
 223: 			if (bDebugReflections)
 224: 				DrawDebugPoint(GetWorld(), hit.HitResult.ImpactPoint, 10.0f, FColor::Red, false, 0.0f);
 225: 			if (bDebugNormals)
 226: 				DrawDebugLine(GetWorld(), hit.HitResult.ImpactPoint, hit.HitResult.ImpactPoint + hit.HitResult.ImpactNormal * 10, FColor::Blue, false, 0.0f, 0, 1.0f);
 227: 		}
 228: 		else {
 229: 			noHitCounter++;
 230: 		}
 231: 	}
 232: 	UE_LOG(Radar, Log, TEXT("SimpleRadar: No reflection from radar: %d of %d"), noHitCounter, Sensor.RecordedHits.Num());
 233: }
